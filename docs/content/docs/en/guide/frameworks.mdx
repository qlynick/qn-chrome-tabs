---
title: Vue and React
description: Set Web Component properties in a framework
---

## Vue

```vue
<script setup>
import { onMounted, ref } from 'vue';
import 'qn-chrome-tabs';

const tabsRef = ref();

onMounted(() => {
  tabsRef.value.tabs = [{ id: 'home', title: 'Home' }];
  tabsRef.value.activeTabId = 'home';
});
</script>

<template>
  <qn-chrome-tabs ref="tabsRef" />
</template>
```

## React

```tsx
import { useEffect, useRef } from 'react';
import 'qn-chrome-tabs';

export function Tabs() {
  const ref = useRef(null);

  useEffect(() => {
    ref.current.tabs = [{ id: 'home', title: 'Home' }];
    ref.current.activeTabId = 'home';
  }, []);

  return <qn-chrome-tabs ref={ref} />;
}
```
