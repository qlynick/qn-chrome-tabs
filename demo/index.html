    Quick Start
    
            
    快速上手
    
          
    
          纯 HTML 可直接引用 Git 构建产物，也可通过包管理器安装。
        
    

        
    
          
    npmnpm install qn-chrome-tabs
    
          
    Yarnyarn add qn-chrome-tabs
    
          
    pnpmpnpm add qn-chrome-tabs
    
          
    Bunbun add qn-chrome-tabs
    
        
    

        
    
          
    
            纯 HTML（无需安装）
            
    <qn-chrome-tabs id="tabs"></qn-chrome-tabs>
<script
  type="module"
  src="https://cdn.jsdelivr.net/gh/qlynick/qn-chrome-tabs@v1.0.1/dist/qn-chrome-tabs.js"
></script>
<script type="module">
  const tabs = document.querySelector('#tabs');
  tabs.tabs = [
    { id: 'home', title: '首页' },
    { id: 'docs', title: '文档' },
  ];
  tabs.activeTabId = 'home';
</script>
    
          
    

          
    
            Vue
            
    <script setup>
import { onMounted, ref } from 'vue';
import 'qn-chrome-tabs';

const tabsRef = ref();
onMounted(() => {
  tabsRef.value.tabs = [
    { id: 'home', title: '首页' },
    { id: 'docs', title: '文档' },
  ];
  tabsRef.value.activeTabId = 'home';
});
</script>

<template>
  <qn-chrome-tabs ref="tabsRef" />
</template>
    
          
    

          
    
            React
            
    import { useEffect, useRef } from 'react';
import 'qn-chrome-tabs';

export function Tabs() {
  const tabsRef = useRef(null);

  useEffect(() => {
    tabsRef.current.tabs = [
      { id: 'home', title: '首页' },
      { id: 'docs', title: '文档' },
    ];
    tabsRef.current.activeTabId = 'home';
  }, []);

  return <qn-chrome-tabs ref={tabsRef} />;
}
